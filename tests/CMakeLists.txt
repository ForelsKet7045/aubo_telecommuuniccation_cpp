cmake_minimum_required(VERSION 3.20)
project(aubo_offline_tests LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use Visual Studio x64 for these offline tests.")
endif()
add_library(mock_sdk SHARED mock_sdk.cpp)
set_target_properties(mock_sdk PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_definitions(mock_sdk PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(mock_sdk PRIVATE ../sdk/include)
target_compile_features(mock_sdk PRIVATE cxx_std_17)
target_compile_options(mock_sdk PRIVATE /utf-8 /W4)
add_executable(robot_communication_mock ../main.cpp)
target_include_directories(robot_communication_mock PRIVATE ../sdk/include)
target_compile_features(robot_communication_mock PRIVATE cxx_std_17)
target_compile_options(robot_communication_mock PRIVATE /utf-8 /W4)
target_link_libraries(robot_communication_mock PRIVATE mock_sdk)
